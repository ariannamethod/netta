"""G1 over the working grid observed in the sitting's own streams.
State = (cnt, freq, factor) of every candidate any of the 15 court streams faced.
log   : log(w + 1e-300) - log(1 + 0.5*freq), math.log = platform libm, as the old mouth
exact : w_a*(2+freq_b) vs w_b*(2+freq_a) in IEEE double, as netta_mouth.c:917-919
truth : the same cross product in Python/NumPy integers (2w is an integer for factor 1 and 1.5)
--rule perturbed swaps exact for a wrong comparator (2.5 in place of 2): the rejecting polarity."""
import math, sys
from collections import defaultdict
from fractions import Fraction
import numpy as np

rule = sys.argv[sys.argv.index("--rule") + 1] if "--rule" in sys.argv else "exact"
dumps = [a for a in sys.argv[1:] if a.endswith(".dump")]
states = set()
for p in dumps:
    for line in open(p):
        f = line.rstrip("\n").split("\t")
        if f[0] == "cand":
            cnt, freq, factor, w = int(f[4]), int(f[5]), float(f[6]), float(f[7])
            assert w == cnt * factor
            states.add((cnt, freq, factor, w))
states = sorted(states)
n = len(states)
cnt = np.array([s[0] for s in states], dtype=np.int64)
fr = np.array([s[1] for s in states], dtype=np.int64)
fac = sorted({s[2] for s in states})
w = np.array([s[3] for s in states], dtype=np.float64)
w2 = np.array([int(2 * s[3]) for s in states], dtype=np.int64)
assert all(2 * s[3] == int(2 * s[3]) for s in states), "2w not integral: truth needs Fractions"
ls = np.array([math.log(s[3] + 1e-300) - math.log(1.0 + 0.5 * s[1]) for s in states])
print("rule under test: %s" % rule)
print("observed states: %d  factors: %s  cnt 1..%d  freq %d..%d" % (n, fac, cnt.max(), fr.min(), fr.max()))
print("ordered pairs: %d" % (n * n - n))
tab = defaultdict(int)
exact_vs_truth = 0
div_in, div_out = 0, 0
out_examples = []
off = 2.5 if rule == "perturbed" else 2.0
for i0 in range(0, n, 400):
    i1 = min(n, i0 + 400)
    A = slice(i0, i1)
    pa = w[A, None] * (off + fr[None, :])
    pb = w[None, :] * (off + fr[A, None])
    ex = np.sign(pa - pb).astype(np.int8) if rule != "log" else None
    lg = np.sign(ls[A, None] - ls[None, :]).astype(np.int8)
    tr = np.sign(w2[A, None] * (2 + fr[None, :]) - w2[None, :] * (2 + fr[A, None])).astype(np.int8)
    if rule == "log":
        ex = lg  # the old comparator as the rule under test
    mask = ~np.eye(n, dtype=bool)[A]
    for a in (-1, 0, 1):
        for b in (-1, 0, 1):
            for c in (-1, 0, 1):
                k = int(np.count_nonzero((lg == a) & (ex == b) & (tr == c) & mask))
                if k: tab[(a, b, c)] += k
    exact_vs_truth += int(np.count_nonzero((ex != tr) & mask))
    d = (lg != ex) & mask
    div_in += int(np.count_nonzero(d & (tr == 0)))
    dout = d & (tr != 0)
    div_out += int(np.count_nonzero(dout))
    if len(out_examples) < 5:
        for r, c in zip(*np.nonzero(dout)):
            out_examples.append((states[i0 + r], states[c]))
            if len(out_examples) >= 5: break
# with rule=log the divergence measure is against the exact cross product, recomputed
if rule == "log":
    div_in = div_out = 0
    for i0 in range(0, n, 400):
        i1 = min(n, i0 + 400); A = slice(i0, i1)
        ex = np.sign(w[A, None] * (2 + fr[None, :]) - w[None, :] * (2 + fr[A, None]))
        lg = np.sign(ls[A, None] - ls[None, :])
        tr = np.sign(w2[A, None] * (2 + fr[None, :]) - w2[None, :] * (2 + fr[A, None]))
        mask = ~np.eye(n, dtype=bool)[A]
        d = (lg != ex) & mask
        div_in += int(np.count_nonzero(d & (tr == 0))); div_out += int(np.count_nonzero(d & (tr != 0)))
print("  log  test truth  count")
for k in sorted(tab):
    print("  %+d   %+d   %+d   %d" % (k[0], k[1], k[2], tab[k]))
print("rule-under-test disagrees with truth: %d" % exact_vs_truth)
print("log vs exact divergences inside an equal class: %d" % div_in)
print("log vs exact divergences outside an equal class: %d" % div_out)
for a, b in out_examples:
    print("  OUTSIDE: a(cnt %d freq %d factor %g) b(cnt %d freq %d factor %g)" % (a[0], a[1], a[2], b[0], b[1], b[2]))
classes = defaultdict(list)
for s in states:
    classes[Fraction(int(2 * s[3]), 2 * (2 + s[1]))].append(s)
multi = {v: m for v, m in classes.items() if len(m) > 1}
def split_by(m, key):
    return len({key(s) for s in m}) > 1
split_log = {v: m for v, m in multi.items() if split_by(m, lambda s: math.log(s[3] + 1e-300) - math.log(1.0 + 0.5 * s[1]))}
split_test = {v: m for v, m in multi.items() if rule == "log" and v in split_log}
print("equal classes (value w/(2+freq)) with >=2 members: %d; split by the log comparator: %d; split by the rule under test: %d" % (
    len(multi), len(split_log), len(split_test) if rule == "log" else 0))
print("classes split by the log comparator (value: members as cnt/freq/factor):")
for v in sorted(split_log):
    print("  %s: %s" % (v, " ".join("%d/%d/%g" % (s[0], s[1], s[2]) for s in split_log[v])))
if rule == "exact":
    ok = exact_vs_truth == 0 and div_out == 0
    print("G1_OBSERVED_RESULT=%s" % ("PASS: every log/exact divergence lies inside a mathematically equal class; exact never disagrees with truth"
                                   if ok else "FAIL: divergence outside an equal class or exact disagrees with truth"))
elif rule == "log":
    ok = len(split_test) > 0
    print("G1_OBSERVED_RED_LOG=%s" % ("detector finds %d classes split by the log comparator" % len(split_test) if ok else "FAIL: detector found no split"))
else:
    ok = exact_vs_truth == 0
    print("G1_OBSERVED_RESULT=%s" % ("PASS" if ok else "FAIL: rule under test disagrees with truth on %d pairs" % exact_vs_truth))
sys.exit(0 if ok else 1)
