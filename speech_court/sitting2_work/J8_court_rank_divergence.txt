import sys
from fractions import Fraction
dump_old, dump_new, trace, step = sys.argv[1], sys.argv[2], sys.argv[3], int(sys.argv[4])
perturb = len(sys.argv) > 5 and sys.argv[5] == "--perturb"

def read(p):
    head, out = None, []
    for line in open(p):
        f = line.rstrip("\n").split("\t")
        if f[0] == "step": head = dict(zip(f[0::2], f[1::2]))
        if f[0] == "rank": out.append((int(f[3]), int(f[5]), float(f[7])))  # tok, cnt, factor
    return head, out

rows = [l.rstrip("\n").split("\t") for l in open(trace)][1:]
win = [int(r[1]) for r in rows[:step]][-12:]   # REP_WINDOW = 12, netta_mouth.c:38
ho, old = read(dump_old)
hn, new = read(dump_new)
limit = int(hn["limit"])
assert int(ho["limit"]) == limit and sorted(old) == sorted(new), "candidate multisets or limit differ"
if perturb:
    new[0], new[1] = new[1], new[0]
    print("PERTURBED: swapped new ranks 0 and 1 (tok %d, tok %d)" % (new[0][0], new[1][0]))
def val(c):  # exact rational w/(2+freq); w = cnt*factor read back exactly from the double
    n, d = (c[1] * c[2]).as_integer_ratio()
    return Fraction(n, d * (2 + win.count(c[0])))
truth = sorted(new, key=lambda c: (-val(c), c[0]))[:limit]
print("candidates=%d limit=%d r=%s tot=%s chosen_at old=%s new=%s chosen old=%s new=%s" % (
    len(new), limit, hn["r"], hn["tot"], ho["chosen_at"], hn["chosen_at"], ho["chosen"], hn["chosen"]))
okA = new[:limit] == truth
print("A new top-%d == exact rational order with lower-id tie-break: %s" % (limit, okA))
diff_slots, bad = 0, []
for i in range(limit):
    a, b = old[i], new[i]
    if a[0] != b[0]:
        diff_slots += 1
        if val(a) != val(b): bad.append((i, a, b))
    print("  slot %2d old tok %4d (cnt %d freq %d) value %-4s | new tok %4d (cnt %d freq %d) value %-4s %s" % (
        i, a[0], a[1], win.count(a[0]), val(a), b[0], b[1], win.count(b[0]), val(b),
        "" if a[0] == b[0] else ("same class" if val(a) == val(b) else "DIFFERENT VALUE")))
print("B slots holding a different token: %d; of them with a different exact value: %d" % (diff_slots, len(bad)))
out_old = sorted(set(c[0] for c in old[:limit]) - set(c[0] for c in new[:limit]))
in_new = sorted(set(c[0] for c in new[:limit]) - set(c[0] for c in old[:limit]))
print("left the top-%d: %s values %s | entered: %s values %s" % (limit,
      out_old, [str(val(c)) for c in old if c[0] in out_old], in_new, [str(val(c)) for c in new if c[0] in in_new]))
ok = okA and not bad
print("G1_STEP_RESULT=%s" % ("PASS: the two comparators differ only inside one mathematically equal class" if ok else
      "FAIL: a slot changed exact value, or the new prefix is not the exact order"))
sys.exit(0 if ok else 1)
